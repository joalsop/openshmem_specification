Signaling Operation
    & \begin{BVerbatim}
shmem_put_signal[_nbi](dest, source, nelems,
sig_addr, signal, sig_op, pe):
  for i in nelems:
    R1 = LD source[i], my_pe
    ST R1 -> dest[i], pe
  switch sig_op:
  case SHMEM_SIGNAL_SET:
    ST signal -> sig_addr, pe
  case SHMEM_SIGNAL_ADD:
    R3 = LD sig_addr, pe
    R2 = R3 + signal
    ST R2 -> sig_addr, pe
    \end{BVerbatim}
